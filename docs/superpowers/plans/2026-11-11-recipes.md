## Shared contracts (every task uses these exact names)

```ts
// lib/types.ts
export type Format = "reel" | "landscape";        // was "reel" | "explainer"; migrate explainer → landscape
export type RecipeId = "explainer" | "product-launch" | "motion-graphic" | "data-story" | "pr-video" | "custom";
export interface VideoOptions {
  tone?: string; length?: string; audience?: string; voice?: string; exactWording?: string; cta?: string;
  angle?: string; palette?: string; voiceTone?: string; musicMood?: string;   // new knobs
}
export interface Video { /* existing fields */ recipe: RecipeId; }
```

```ts
// lib/recipes.ts
export interface Recipe {
  id: RecipeId;
  label: string;               // "Explainer"
  tagline: string;             // one line for the picker card
  inputPlaceholder: string;    // textarea placeholder for this type
  narration: boolean;          // false → spoken lines may be empty; no voice stage
  scenes: { min: number; max: number };
  duration: { reel: string; landscape: string };  // e.g. "~30 s", "about 45–60 s"
  angles: string[];            // may be empty
  knobs: Array<"angle" | "palette" | "voiceTone" | "musicMood" | "length" | "audience" | "exactWording" | "cta" | "voice">;
  skillRefs: string[];         // paths relative to the hyperframes skill dir, e.g. "references/types/explainer.md"
  needsUrl: boolean;           // product-launch and pr-video want a URL in the input
}
export const RECIPE_IDS: readonly RecipeId[];
export function listRecipes(): Recipe[];                 // ordered as RECIPE_IDS
export function getRecipe(id: string): Recipe;           // throws on unknown id
export function recipeText(id: RecipeId, part: "script" | "build"): string; // contents of recipes/<id>/<part>.md
```

The files are `studio/recipes/<id>/recipe.json` (validated with zod at load), `script.md` (story shape and scene rules for the script writer) and `build.md` (look, audio, caption and technique defaults for the build agent).

```ts
// lib/script-schema.ts additions
look: { ground: string; ink: string; accent: string;          // hex like "#1D5AA6"
        type: string;                                         // type direction, e.g. "condensed bold sans, 9% frame height titles"
        captionStyle: string;                                 // a catalog style by tone or "none"
        voiceTone: string; musicMood: string }               // "calm male, unhurried"; "low pulsing synth under -18 dB" or "none"
concept: { device: string; form: string; spectacle: string }  // from the skill's story.md device/form tables + one spectacle beat
scenes[i]: + motion?: string; transition?: string; block?: string   // optional
validateScript(raw: unknown, ctx: { format: Format; recipe: Recipe }):
  - scene count within recipe.scenes
  - when recipe.narration: spoken non-empty and ≤ 19 words; when not: spoken must be "" (empty)
  - onScreen limits by format: reel ≤ 6 words, landscape ≤ 10; onScreen ≠ spoken when spoken non-empty
  - look.ground/ink/accent must be #RGB or #RRGGBB
```
